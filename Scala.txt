nano bubblesort.scala


// Bubble Sort using Scala (with Spark context just for setup)

val data = Array(64, 34, 25, 12, 22, 11, 90)

println("Original Array:")
println(data.mkString(" "))

// Bubble Sort logic
for (i <- 0 until data.length) {
  for (j <- 0 until data.length - i - 1) {
    if (data(j) > data(j + 1)) {
      val temp = data(j)
      data(j) = data(j + 1)
      data(j + 1) = temp
    }
  }
}

println("Sorted Array:")
println(data.mkString(" "))
